# Run a command with its whole output in $LOG, then print its exit status.
# Its stdin is /dev/null: Babel hands the block's body to the shell on

# The run's verdict: each failing test's name and the tally.
verdict() {
  grep -E '^(FAIL|ERROR) in |^Ran |failures, ' "$LOG" | sed -E 's/ \([a-z_]+\.clj:[0-9]+\)$//'

# Every line a pristine-output scan flags, timestamps normalized. Left out:
# compile and tally lines, lein test's per-namespace headers (one namespace
# is named js-runtime-warning-test), a failed assertion's expected and
# actual lines, which verdict already reports and whose printed sets have
# no stable order, and port 9630's warning, which depends on what else runs
# on the machine and which the plan's Global Constraints name.
scan() {
  grep -i -E 'warn|error|exception|traceback' "$LOG" \
    | grep -v -E '^\[:[a-z-]+\] Build completed\.|^[0-9]+ failures, [0-9]+ errors\.$|^lein test |^ *(expected|actual): |TCP Port 9630 in use' \
    | sed -E 's/^\[[0-9-]+ [0-9:.]+ - /[<time> - /' \
    | sort | uniq -c | sed -E 's/^ +//'
  echo "scan done"
}

(defn- shadow-nrepl-port []
  (parse-long (str/trim (slurp ".shadow-cljs/nrepl.port"))))

(defn- script-config
  "A :node-script build whose output carries the REPL client."
  [build-id]
  {:build-id build-id
   :target :node-script
   :main 'shadow.cljs.js-runtime-watch-probe/main
   :output-to (str "target/" (name build-id) "/script.js")
   :js-runtime :bun})

(defn- eval-after-select
  "Clone a session on shadow's nREPL, select BUILD-ID with nrepl-select, then
  evaluate CODE as ClojureScript. Returns CODE's :out and :err."
  [build-id code]
  (with-open [conn (nrepl/connect :port (shadow-nrepl-port))]
    (let [session (nrepl/client-session (nrepl/client conn 60000))
          run #(nrepl/combine-responses (nrepl/message session {:op "eval" :code %}))]
      (run (format "(shadow.cljs.devtools.api/nrepl-select %s)" build-id))
      (select-keys (run code) [:out :err]))))

(defn- timed
  "[result elapsed-ms] of calling F."
  [f]
  (let [start (System/currentTimeMillis)
        result (f)]
    [result (- (System/currentTimeMillis) start)]))

(defn- poll
  "The first truthy value of F within TIMEOUT-MS, or nil."
  [timeout-ms f]
  (let [deadline (+ (System/currentTimeMillis) timeout-ms)]
    (loop []
      (or (f)
          (when (< (System/currentTimeMillis) deadline)
            (Thread/sleep 50)
            (recur))))))

(defn- await-runtime
  "The first REPL runtime connected to BUILD-ID within TIMEOUT-MS, or nil."
  [build-id timeout-ms]
  (poll timeout-ms #(->> (api/repl-runtimes build-id) (map :client-id) first)))

(defn- when-bun
  "Calls F, or passes vacuously when bun is not installed."
  [f]
  (if-not (bun-available?)
    (testing "Bun not installed"
      (is true))
    (f)))

(defn- with-watched-build
  "Watch CONFIG, compiled once, call F with it, and stop the worker. CLI-OPTS
  go to api/watch. Without bun, F does not run and the test passes
  vacuously."
  ([config f] (with-watched-build config {} f))
  ([{:keys [build-id] :as config} cli-opts f]
   (when-bun
     (fn []
       (try
         ;; a worker left by an aborted run would ignore CLI-OPTS; failing
         ;; here still stops it, so the next run starts clean
         (assert (= :watching (api/watch config (merge {:autobuild false :sync true} cli-opts)))
                 (str build-id " was already being watched"))
         (f config)
         (finally
           (api/stop-worker build-id)))))))

(def ^:private no-runtime-message
  "No available JS runtime.\nSee https://shadow-cljs.github.io/docs/UsersGuide.html#repl-troubleshooting")

(def ^:private connect-window-ms
  "Long enough for a process started at the end of the compile to connect: a
  cold bun start loading a dev build takes about a second."
  5000)

(def ^:private bun-check "(println (some? js/process.versions.bun))")

(deftest a-watched-build-starts-no-process
  ;; --js-runtime reaches the build through :config-merge, so the raw build
  ;; config never carries it; both ways of choosing the runtime are checked
  (doseq [[label config cli-opts]
          [["from the build config" (script-config :js-runtime-watch-config) {}]
           ["from the command line" (dissoc (script-config :js-runtime-watch-cli) :js-runtime)
            {:config-merge [{:js-runtime :bun}]}]]]
    (testing label
      (with-watched-build config cli-opts
        (fn [{:keys [build-id]}]
          (is (nil? (await-runtime build-id connect-window-ms))
              "no runtime connected after the compile"))))))

(deftest an-eval-without-a-runtime-fails-at-once
  (with-watched-build (script-config :js-runtime-watch-eval)
    (fn [{:keys [build-id]}]
      (let [[result ms] (timed #(eval-after-select build-id "(println :unreachable)"))]
        (is (= {:err no-runtime-message} result))
        (is (< ms 2000) "answered without waiting for a runtime")))))

(deftest an-eval-during-the-first-compile-gets-no-runtime
  ;; A tool may select the build and evaluate as soon as the watch's nREPL
  ;; port is written, while the first compile still runs. Nothing in the
  ;; fork waits for a runtime on the tool's behalf: it polls
  ;; api/repl-runtimes itself. Every other test here watches with :sync
  ;; true, which hides this; deleting the build's cache keeps the compile
  ;; slow enough to evaluate during it. The compile is checked unfinished
  ;; after the answer, so an eval that waited for it fails here; a compile
  ;; that outran the eval fails too, never passes.
  (let [build-id :js-runtime-watch-first-compile]
    (doseq [f (reverse (file-seq (io/file (get-in (runtime/get-instance!) [:config :cache-root])
                                          "builds" (name build-id))))]
      (io/delete-file f true))
    (with-watched-build (script-config build-id) {:sync false}
      (fn [_]
        (is (= {:err no-runtime-message} (eval-after-select build-id "(println :unreachable)")))
        (is (nil? (-> (api/get-worker build-id) :state-ref deref :last-build-provides))
            "the answer came before the first compile ended")))))

(deftest a-runtime-started-on-the-output-is-evaluated-in
  ;; the contract a launcher relies on: run the output, and the REPL
  ;; evaluates in what connects
  (with-watched-build (script-config :js-runtime-watch-by-hand)
    (fn [{:keys [build-id output-to]}]
      (let [process (-> (ProcessBuilder. ^java.util.List ["bun" output-to])
                        (.redirectErrorStream true)
                        (.redirectOutput java.lang.ProcessBuilder$Redirect/DISCARD)
                        (.start))]
        (try
          (is (some? (await-runtime build-id 10000)) "the process's client connected")
          (is (= {:out "true\n"} (eval-after-select build-id bun-check)))
          (finally
            (.destroyForcibly process)
            (.waitFor process)))))))

(deftest watching-a-running-build-by-its-config-says-so
  ;; with-watched-build relies on this to catch a worker left by an
  ;; aborted run
  (with-watched-build (script-config :js-runtime-watch-rewatch)
    (fn [config]
      (is (= :already-watching (api/watch config {:autobuild false}))))))

(deftest node-repl-evaluates-under-the-selected-runtime
  ;; Options as the CLI's --js-runtime builds them.
  (when-bun
    #(let [build-id :js-runtime-watch-node-repl]

         {:build-id build-id :js-runtime :bun :config-merge [{:js-runtime :bun}]})
       (try
         (is (some? (await-runtime build-id 10000)) "its own process connects")
         (is (= ["true"] (:results (api/cljs-eval build-id "(some? js/process.versions.bun)" {})))
             "and evaluates under bun")
         (finally
           (api/stop-worker build-id))))))
CLJ
